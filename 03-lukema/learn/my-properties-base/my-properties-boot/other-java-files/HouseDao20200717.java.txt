    private static void calculateMonthlyPayment() {
        HOUSE_LIST.forEach((house) -> {
            float monthlyRate = house.getInterestRate() / 100 / 12;
            Date maturityDate = null;
            try {
                maturityDate = new SimpleDateFormat(DATE_FORMAT).parse(house.getMaturityDate());
            } catch (ParseException e) {
                log.error(() -> "Unable to format maturity date.", e);
            }

            Date dateUpdated = null;
            try {
                dateUpdated = new SimpleDateFormat(DATE_FORMAT).parse(DATE_UPDATED);
            } catch (ParseException e) {
                log.error(() -> "Unable to format update date.", e);
            }

            if (!(maturityDate == null || dateUpdated == null)) {

                LocalDate startDateInclusive = dateUpdated.toInstant().atZone(ZoneId.systemDefault()).toLocalDate();
                LocalDate endDateExclusive = maturityDate.toInstant().atZone(ZoneId.systemDefault()).toLocalDate();

                // LOG.info(startDateInclusive + " " + endDateExclusive);

                Period age = Period.between(startDateInclusive, endDateExclusive);

                int numberOfMonthToMaturity = age.getYears() * 12 + age.getMonths();

                float monthlyPayment = (float) (monthlyRate / (1 - Math.pow((1 + monthlyRate), -1.0 * numberOfMonthToMaturity)))
                        * house.getInitialPrincipalBalance();

                float monthlyInterest = house.getInitialPrincipalBalance() * monthlyRate;
                float monthlyPrincipal = monthlyPayment - monthlyInterest;

                log.info(() -> "Item: " + house.getId() + " numberOfMonthToMaturity: " + numberOfMonthToMaturity + " Monthly Payment: "
                        + (int) monthlyPayment + " principal: " + (int) monthlyPrincipal + " interest: " + (int) monthlyInterest);

                house.setMonthlyInterestPayment(monthlyInterest);
                house.setMonthlyPrincipalPayment(monthlyPrincipal);

                house.setCalculatedMonthlyEscrow(monthlyPayment + house.getMonthlyCountyPropertyTax() + house.getMonthlyLIDTax()
                        + house.getMonthlyMUDTax() + house.getMonthlyInsurancePayment());
            }
        });
    }

                return house;
            }
        }

        return null;
    }
