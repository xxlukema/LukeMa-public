

Use Cygwin. openssl must be installed.

1. Generate a private key

   openssl genrsa -out luke-private.pem 2048
   
2. Derive the public key

   openssl rsa -in luke-private.pem -pubout  > luke-public.pem

3. Create a self-signed X509 certificate

   openssl req -new -x509 -key luke-private.pem -out luke-certificate.pem -days 365
   Common name: localhost or 127.0.0.1 or 10.233.12.89
   
4. Create a PKCS12 keystore and import the host certificate we just created

   openssl pkcs12 -export -out luke-keystore.pkcs12 -in luke-certificate.pem -inkey luke-private.pem
   Password: lukema

5. Convert the PKCS12 keystore to Java keystore using Java keytool

   keytool -importkeystore -srckeystore luke-keystore.pkcs12 -srcstoretype PKCS12 -destkeystore luke-keystore.jks -deststoretype JKS
   Password: lukema

6. List keystore

   keytool -list -v -keystore luke-keystore.jks -storepass lukema
   
7. Rename alias

   keytool -changealias -alias "1" -destalias "luke" -keypass lukema -keystore luke-keystore.jks -storepass lukema
   
8. Configure Tomcat HTTPS Connector

   <!-- Tomcat 8.0 -->
   <Connector port="8443" protocol="org.apache.coyote.http11.Http11NioProtocol" 
              SSLEnabled="true" maxThreads="150" secure="true" scheme="https" keyAlias="luke" 
			  keystoreFile="D:/JavaCAPS/appserver/domains/MyRest/config/luke-keystore.jks" 
			  keystorePass="lukema" 
			  clientAuth="false" sslProtocol="TLS" />

	<!-- Tomcat 8.5 -->
   	<Connector port="8443" protocol="org.apache.coyote.http11.Http11NioProtocol"
               maxThreads="150" SSLEnabled="true">
        <SSLHostConfig>
            <Certificate certificateKeystoreFile="D:/JavaCAPS/appserver/domains/MyRest/config/luke-keystore.jks"
			             certificateKeystorePassword="lukema"
                         type="RSA" />
        </SSLHostConfig>
    </Connector>
